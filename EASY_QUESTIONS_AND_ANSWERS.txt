================================================================================

query = """

QUESTION 27: Customers by age group
================================================================================

query = """
SELECT 
    CASE 
        WHEN EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM date_of_birth) < 30 THEN '18-29'
        WHEN EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM date_of_birth) < 40 THEN '30-39'
        WHEN EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM date_of_birth) < 50 THEN '40-49'
        WHEN EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM date_of_birth) < 60 THEN '50-59'
        WHEN EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM date_of_birth) < 70 THEN '60-69'
        ELSE '70+'
    END as age_group,
    COUNT(*) as count
FROM customers
GROUP BY age_group
ORDER BY age_group
"""
con.execute(query).df()

    SUM(CASE WHEN gender = 'M' THEN 1 ELSE 0 END) as males,

TIPS:
-----
1. Always connect to database first: con = duckdb.connect('data.duckdb', read_only=True)
2. Use single quotes for text in SQL: WHERE occupation = 'Retired'
3. Use COUNT(*) to count rows
4. Use AVG(), MAX(), MIN() for calculations
5. Use GROUP BY when counting categories
6. Use ORDER BY to sort results
7. Use LIMIT to show only top results
